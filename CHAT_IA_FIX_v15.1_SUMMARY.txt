━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Découplage complet entre UI (affichage) et Memory (sauvegarde) :
• useEffect charge l'historique UNIQUEMENT au changement de mode
• saveMessage() est silencieux (pas de re-render)
• Protection anti-duplication ajoutée
• Garde-fous de vérification post-réponse
