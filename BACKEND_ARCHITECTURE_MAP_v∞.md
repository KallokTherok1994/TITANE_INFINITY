// 1. Loader robuste avec fallback
pub fn load_identity_matrix() -> IdentityMatrix {
    match fs::read_to_string("identity.json") {
        Ok(content) => {
            match serde_json::from_str::<IdentityMatrix>(&content) {
                Ok(matrix) => {
                    if validate_matrix(&matrix) {
                        matrix
                    } else {
                        log::warn!("Identity matrix invalid, using default");
                        DEFAULT_IDENTITY_MATRIX.clone()
                    }
                }
                Err(e) => {
                    log::error!("Failed to parse identity.json: {}", e);
                    DEFAULT_IDENTITY_MATRIX.clone()
                }
            }
        }
        Err(_) => {
            log::info!("identity.json not found, creating default");
            DEFAULT_IDENTITY_MATRIX.clone()
        }
    }
}

// 2. Atomic write pattern
pub fn save_identity_matrix(matrix: &IdentityMatrix) -> Result<(), AppError> {
    let content = serde_json::to_string_pretty(matrix)?;
    let temp_path = "identity.json.tmp";
    let final_path = "identity.json";
    let backup_path = "identity.json.backup";

    // Backup existant
    if Path::new(final_path).exists() {
        fs::copy(final_path, backup_path)?;
    }

    // Write to temp
    fs::write(temp_path, content)?;

    // Atomic rename
    fs::rename(temp_path, final_path)?;

    Ok(())
}

// 3. Migration versionnée
pub fn migrate_identity_matrix(matrix: &mut IdentityMatrix) {
    if matrix.version < "v2.0" {
        // Add new fields, transform old structure
        matrix.version = "v2.0";
    }
}
