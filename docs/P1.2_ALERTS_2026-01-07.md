```typescript
export enum AlertType {
  LATENCY_THRESHOLD = 'latency_threshold',      // P95/avg exceeded
  ERROR_RATE = 'error_rate',                     // High error rate
  HEALTH_DEGRADATION = 'health_degradation',     // System health drop
```

```typescript
export enum AlertSeverity {
  INFO = 'info',           // Informational
  WARNING = 'warning',     // Needs attention
  CRITICAL = 'critical'    // Requires immediate action
}
```

#### C. Alert Structure
```typescript
interface PerformanceAlert {
  id: string;              // Unique alert ID
  timestamp: number;       // Unix timestamp
  severity: AlertSeverity; // INFO | WARNING | CRITICAL
  type: AlertType;         // Alert type
  metricName: string;      // Metric that triggered
  message: string;         // Human-readable message
  value: number;           // Actual value
  threshold: number;       // Threshold exceeded
  metadata?: Record<string, any>; // Additional data
}
```

### 3. Alert Manager API ✅ COMPLETE

#### A. Monitoring Control
```typescript
class PerformanceAlertManager {
  // Start monitoring
  start(intervalMs: number = 60000): void

  // Stop monitoring
  stop(): void

  // Add custom threshold
  addThreshold(id: string, threshold: AlertThreshold): void

  // Remove threshold
  removeThreshold(id: string): boolean

  // Update threshold
  updateThreshold(id: string, updates: Partial<AlertThreshold>): boolean
}
```

**Example Usage:**
```typescript
// Start monitoring (check every 60 seconds)
performanceAlerts.start(60000);

// Add custom threshold for database queries
performanceAlerts.addThreshold('db-queries', {
  metricPattern: /^database\.query/,
  p95Threshold: 300,      // 300ms max
  avgThreshold: 150,      // 150ms average
  severity: AlertSeverity.WARNING,
  enabled: true
});

// Update existing threshold
performanceAlerts.updateThreshold('db-queries', {
  p95Threshold: 200      // Tighten to 200ms
});

// Stop monitoring
performanceAlerts.stop();
```

#### B. Alert Handlers
```typescript
// Register custom alert handler
performanceAlerts.onAlert((alert) => {
  if (alert.severity === AlertSeverity.CRITICAL) {
    // Send to monitoring service
    sendToDatadog(alert);

    // Send Slack notification
    notifySlack({
      channel: '#performance-alerts',
      text: `🚨 Critical: ${alert.message}`
    });

    // Log to file
    fs.appendFileSync('alerts.log', JSON.stringify(alert) + '\n');
  }
});

// Unregister handler
performanceAlerts.offAlert(handler);
```

```typescript
// Slack integration
performanceAlerts.onAlert(async (alert) => {

    await fetch('https://hooks.slack.com/services/YOUR/WEBHOOK/URL', {

            text: {

            }
          }
        ]
      })
    });
  }
});

// Datadog integration
performanceAlerts.onAlert(async (alert) => {
  await fetch('https://api.datadoghq.com/api/v1/events', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'DD-API-KEY': process.env.DATADOG_API_KEY
    },
    body: JSON.stringify({
      title: `Performance Alert: ${alert.metricName}`,
      text: alert.message,
      priority: alert.severity === AlertSeverity.CRITICAL ? 'high' : 'normal',
      tags: [
        `severity:${alert.severity}`,
        `type:${alert.type}`,
        `metric:${alert.metricName}`
      ]
    })
  });
});
```
