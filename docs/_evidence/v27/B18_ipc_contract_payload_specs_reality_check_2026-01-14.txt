    74	fn mask_secret_for_display(secret: &str) -> String {
    75	    if secret.is_empty() {
    76	        return String::new();
    77	    }
    78	
    79	    let mut visible: Vec<char> = secret.chars().rev().take(4).collect();
    80	    visible.reverse();
    81	
    82	    let total = secret.chars().count();
    83	    let masked_len = total.saturating_sub(visible.len());
    84	
    85	    let mut output = String::with_capacity(total);
    86	    if masked_len > 0 {
    87	        output.push_str(&"•".repeat(masked_len));
    88	    }
    89	    for ch in visible {
    90	        output.push(ch);
    91	    }
    92	
    93	    output
    94	}
