// Pipeline requête avec temporalité
rate_limiter.acquire_permit().await?;
breaker.allow_request().await?;

if let Some(cached) = cache.get(&key).await {
    return cached; // Hit
}

let decision = router.route(&request, &registry).await;
// → Provider optimal selon temporalité

let response = execute_api(decision.provider, &request).await?;
cache.set(key, response, endpoint).await;
breaker.record_success().await;
